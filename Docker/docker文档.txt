cat app.py 
from flask import Flask    #引用flask
app = Flask(__name__)

@app.route('/')
def hello():
    return 'Hello, Docker!'

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)

cat requirements.txt 
flask==2.3.0



cat Dockerfile 
#基于debin的python3.10环境，约150M
FROM python:3.10-slim 
#会有警告
ENV USID 601 
ENV USNAME=sfm
ENV APPDIR=/${USNAME}
#切换用户为root（默认为root）
USER root 
RUN mkdir /sfm/; \ 
    groupadd -g ${USID} ${USNAME}; \
    useradd -m -u ${USID} -g ${USNAME} -s /bin/bash ${USNAME}; 
#指定工作目录，不存在则创建。实际效果类似cd /sfm    
WORKDIR /sfm      
COPY --chown=sfm:sfm ./readme.txt ${APPDIR}/README.TXT  
COPY requirements.txt .
COPY app.py .
#复制当前目录下所有文件
#COPY . .                                   
RUN chown -R ${USNAME}:${USNAME} ${APPDIR}  
RUN pip install -r requirements.txt         
ENV redispasswd="C1bsfm_ACB"
USER sfm
ADD ./filename.txt /sfm/                  
ADD --chown=sfm:sfm ./tarfile.tar ${APPDIR}/ 
EXPOSE 5000                                 
CMD ["python", "app.py"] 





#构建镜像
docker build -t zxfbj-demo:latest .
