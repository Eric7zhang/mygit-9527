cat app.py 
from flask import Flask    #引用flask
app = Flask(__name__)

@app.route('/')
def hello():
    return 'Hello, Docker!'

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)


cat Dockerfile 
FROM python:3.10-slim
WORKDIR /sfm
COPY requirements.txt .
RUN pip install -r requirements.txt
COPY . .
EXPOSE 19201
CMD ["python", "app.py"]
